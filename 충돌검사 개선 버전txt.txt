// Check if two sets of cells overlap
bool GameWorld::ShareSameCell(const std::vector<XMINT3>& a, const std::vector<XMINT3>& b) const
{
    for (const auto& cellA : a)
        for (const auto& cellB : b)
            if (cellA.x == cellB.x && cellA.y == cellB.y && cellA.z == cellB.z)
                return true;
    return false;
}

// Compute all grid cells an OBB overlaps
void GameWorld::Compute_CellList_From_OBB(const std::shared_ptr<BoundingOrientedBox>& obb, std::vector<XMINT3>& out_cells) const
{
    out_cells.clear();
    if (!obb) return;

    XMINT3 minCell, maxCell;
    Compute_CellBounds_From_OBB(obb, minCell, maxCell);

    for (int x = minCell.x; x <= maxCell.x; ++x)
        for (int y = minCell.y; y <= maxCell.y; ++y)
            for (int z = minCell.z; z <= maxCell.z; ++z)
                out_cells.emplace_back(XMINT3(x, y, z));
}

// Apply collision response to player when intersecting another OBB
void GameWorld::ResolveCollision(std::shared_ptr<Player>& player, const BoundingOrientedBox& otherOBB)
{
    auto& playerOBB = *player->Get_Collider_OBB();

    if (playerOBB.Intersects(otherOBB))
    {
        XMVECTOR dir = XMLoadFloat3(&playerOBB.Center) - XMLoadFloat3(&otherOBB.Center);
        if (!XMVector3Equal(dir, XMVectorZero()))
        {
            dir = XMVector3Normalize(dir);
            XMVECTOR offset = XMVectorScale(dir, 2.0f); // Push strength
            offset = XMVectorSet(XMVectorGetX(offset), 0, XMVectorGetZ(offset), 0);

            XMStoreFloat3(&playerOBB.Center, XMLoadFloat3(&playerOBB.Center) + offset);
            player->Set_Collider_OBB_Center(playerOBB.Center);
            player->need_to_client_sync = true;
        }
    }
}

// Main collision dispatcher
void GameWorld::Update_Collision_All(const std::array<std::shared_ptr<Player>, MaxPlayer>& players)
{
    // Precompute player cell lists
    std::array<std::vector<XMINT3>, MaxPlayer> player_cells;

    for (int i = 0; i < MaxPlayer; ++i)
    {
        if (players[i] && players[i]->Get_Collider_OBB())
        {
            players[i]->UpdateWorldOBB();
            Compute_CellList_From_OBB(players[i]->Get_Collider_OBB(), player_cells[i]);
        }
    }

    // Step 1: Player vs Fixed Object
    for (int i = 0; i < MaxPlayer; ++i)
    {
        auto& player = players[i];
        if (!player || !player->Get_Collider_OBB()) continue;

        std::unordered_set<const BoundingOrientedBox*> checked;

        for (const auto& cell : player_cells[i])
        {
            auto it = uniform_cell_map.find(cell);
            if (it == uniform_cell_map.end()) continue;

            for (int objIndex : it->second)
            {
                auto& fixed = fixed_object_list[objIndex];
                if (!fixed || !fixed->Get_Collider_OBB()) continue;

                auto& otherOBB = *fixed->Get_Collider_OBB();
                if (checked.count(&otherOBB)) continue;
                checked.insert(&otherOBB);

                ResolveCollision(player, otherOBB);
            }
        }
    }

    // Step 2: Player vs Player
    for (int i = 0; i < MaxPlayer; ++i)
    {
        auto& playerA = players[i];
        if (!playerA || !playerA->Get_Collider_OBB()) continue;

        for (int j = i + 1; j < MaxPlayer; ++j)
        {
            auto& playerB = players[j];
            if (!playerB || !playerB->Get_Collider_OBB()) continue;

            if (ShareSameCell(player_cells[i], player_cells[j]))
            {
                ResolveCollision(playerA, *playerB->Get_Collider_OBB());
                ResolveCollision(playerB, *playerA->Get_Collider_OBB());
            }
        }
    }
}
